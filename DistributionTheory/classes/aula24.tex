\subsection{Apêndice: Continuação das Propriedades da Convolução}
Recordemos as propriedades da convolução de distribuições quando pelo menos um dos fatores possui suporte compacto.

\begin{enumerate}
 \item A delta é o elemento neutro:
 \[
  \delta*u=u*\delta=u,
  \qquad \forall u\in\mathcal D'(\mathbb R^n).
 \]

 \item A convolução é associativa:
 \[
  u_1*(u_2*u_3)=(u_1*u_2)*u_3,
 \]
 sempre que ao menos dois fatores tenham suporte compacto. Em particular, a convolução fornece uma operação interna em $\mathcal E'$.

 \item A convolução é comutativa:
 \[
  u_1*u_2=u_2*u_1.
 \]

 \item A derivação é um operador do tipo convolução:
 \[
  \partial^\alpha(u_1*u_2)
  =(\partial^\alpha u_1)*u_2
  =u_1*(\partial^\alpha u_2).
 \]
 Em particular,
 \[
  \partial^\alpha u
  =\partial^\alpha(\delta*u)
  =(\partial^\alpha\delta)*u,
 \]
 e, para um ODPCC $P(D)$,
 \[
  P(D)u=(P(D)\delta)*u.
 \]

 \item O suporte satisfaz
 \[
  \operatorname{supp}(u_1*u_2)
  \subseteq\operatorname{supp}(u_1)+\operatorname{supp}(u_2).
 \]
 Isso segue, por exemplo, do lema: se $u_j\to u$ em $\mathcal D'$ e $\operatorname{supp}(u_j)\subseteq F_j$, onde $(F_j)$ é uma sequência decrescente de fechados com $F=\bigcap_jF_j$, então
 \[
  \operatorname{supp}(u)\subseteq F.
 \]

 \item Se $u_1$ ou $u_2$ é uma função $\mathcal C^\infty$, as definições antiga e nova de convolução coincidem.
\end{enumerate}

\subsection{Existência e Unicidade de Soluções para ODPCC}
Tendo definido a convolução de duas distribuições quando uma delas possui suporte compacto, podemos usá-la no estudo da equação não homogênea
\[
 P(D)u=v,
\]
onde $P(D)$ é um operador diferencial parcial linear com coeficientes constantes e $E$ é uma solução fundamental de $P(D)$.

A aplicação
\[
 \mathcal D'\times\mathcal E'\longrightarrow\mathcal D',
 \qquad (u,v)\longmapsto u*v
\]
é bilinear.

\begin{prop*}[Existência]
 Se $v\in\mathcal E'(\mathbb R^n)$ e $E$ é uma solução fundamental de $P(D)$, então
 \[
  u:=E*v
 \]
 está bem definido e satisfaz
 \[
  P(D)u=v.
 \]
\end{prop*}

\begin{proof*}
 Com efeito,
 \begin{align*}
  P(D)u
  &=\sum_{|\alpha|\leq m}a_\alpha(\partial^\alpha E)*v\\
  &=\left(\sum_{|\alpha|\leq m}a_\alpha\partial^\alpha E\right)*v\\
  &=(P(D)E)*v
  =\delta*v=v.
 \end{align*}
 \qedsymbol
\end{proof*}

\begin{prop*}[Unicidade entre soluções com suporte compacto]
 Seja $v\in\mathcal E'$. Se $u\in\mathcal E'$ satisfaz $P(D)u=v$, então
 \[
  u=E*v.
 \]
 Em particular, existe no máximo uma solução de suporte compacto de $P(D)u=v$.
\end{prop*}

\begin{proof*}
 Como $u\in\mathcal E'$, podemos associar convolutivamente:
 \[
  u=\delta*u=(P(D)E)*u
  =E*(P(D)u)=E*v.
 \]
 \qedsymbol
\end{proof*}

 A mesma construção mostra uma forma de continuidade em relação aos dados iniciais: se $f_j\to f$ em $\mathcal E'$, então
 \[
  u_j=E*f_j\longrightarrow E*f=u
 \]
 em $\mathcal D'$. Para dados suaves de suporte compacto, a regularização do lado direito se transfere às soluções pela convolução.
\end{tcolorbox}

\subsection{Regularidade das Soluções e Hipoelipticidade}
Queremos estudar a regularidade local das soluções de
\[
 P(x,D)u=f
\]
em um aberto $\Omega\subseteq\mathbb R^n$, quando $f$ é $\mathcal C^\infty$ numa vizinhança de um ponto, sem exigir que $f$ tenha suporte compacto.

\begin{def*}
 Seja $\Omega\subseteq\mathbb R^n$ aberto e
 \[
  P(x,D)=\sum_{|\alpha|\leq m}a_\alpha(x)\partial^\alpha
 \]
 um operador diferencial com coeficientes suaves. Dizemos que $P(x,D)$ é \textbf{hipoelíptico em $\Omega$} se, para toda $u\in\mathcal D'(\Omega)$,
 \[
  \operatorname{supp\,sing}(P(x,D)u)
  =\operatorname{supp\,sing}(u).
 \]
\end{def*}

Sempre temos
\[
 \operatorname{supp\,sing}(P(x,D)u)
 \subseteq\operatorname{supp\,sing}(u),
\]
pois, se $u$ é $\mathcal C^\infty$ numa vizinhança, então todas as suas derivadas e os produtos pelos coeficientes suaves também são $\mathcal C^\infty$ ali. Portanto, a hipoelipticidade equivale à inclusão inversa
\[
 \operatorname{supp\,sing}(u)
 \subseteq\operatorname{supp\,sing}(P(x,D)u).
\]
Equivalentemente, para todo aberto $U\subseteq\Omega$,
\[
 P(x,D)u\in\mathcal C^\infty(U)
 \quad\Longrightarrow\quad
 u\in\mathcal C^\infty(U).
\]
Em palavras, $P(x,D)$ recupera a regularidade de $u$ a partir da regularidade de sua imagem.

\subsubsection*{Exemplos}
\begin{enumerate}
 \item O operador $P(D)=d/dx$ é hipoelíptico em $\mathbb R$. Se
 \[
  \frac{du}{dx}=f\in\mathcal C^\infty,
 \]
 então, escolhendo uma primitiva suave
 \[
  v(x)=\int_a^x f(t)\,dt,
 \]
 temos $(u-v)'=0$, logo $u-v$ é uma constante e $u\in\mathcal C^\infty$.

 Da mesma forma, $d/dx-a$, $a\in\mathbb C$, é hipoelíptico em $\mathbb R$.

 \item O operador $\partial/\partial x$ não é hipoelíptico em $\mathbb R^2$. Por exemplo,
 \[
  u(x,y)=|y|
 \]
 satisfaz $\partial_xu=0\in\mathcal C^\infty$, mas $u\notin\mathcal C^\infty$.

 Analogamente, $\partial_x-\partial_y$ não é hipoelíptico em $\mathbb R^2$, pois funções singulares constantes ao longo das direções características fornecem contraexemplos.

 \item O operador da onda
 \[
  \Box=\frac{\partial^2}{\partial t^2}-\frac{\partial^2}{\partial x^2}
 \]
 não é hipoelíptico. De fato, $u(x,t)=f(x+t)$ satisfaz $\Box u=0$ para qualquer $f$ suficientemente diferenciável, e podemos escolher $f$ não suave.
\end{enumerate}

Por outro lado, $\Delta$, $\partial/\partial\overline z$ e $\partial_t-\Delta_x$ serão hipoelípticos, como consequência do resultado seguinte.

\begin{theorem*}
 Seja
 \[
  P(D)=\sum_{|\alpha|\leq m}a_\alpha\partial^\alpha
 \]
 um ODPCC. Se $P(D)$ possui uma solução fundamental
 \[
  E\in\mathcal C^\infty(\mathbb R^n\setminus\{0\}),
 \]
 então $P(D)$ é hipoelíptico em todo aberto $\Omega\subseteq\mathbb R^n$.
\end{theorem*}

\begin{tcolorbox}[

 A recíproca também é imediata: se $P(D)$ é hipoelíptico e $E$ é uma solução fundamental, então
 \[
  \operatorname{supp\,sing}(E)
  =\operatorname{supp\,sing}(P(D)E)
  =\operatorname{supp\,sing}(\delta)=\{0\},
 \]
 logo $E\in\mathcal C^\infty(\mathbb R^n\setminus\{0\})$.

 A existência de uma solução fundamental para todo ODPCC é garantida pelo Teorema de Malgrange--Ehrenpreis.
\end{tcolorbox}

\begin{proof*}
 Sejam $u\in\mathcal D'(\Omega)$ e $U\subseteq\Omega$ aberto tais que
 \[
  P(D)u=f\in\mathcal C^\infty(U).
 \]
 Mostraremos que $u$ é suave em uma vizinhança de cada $x_0\in U$.

 Escolha $V\Subset U$ uma vizinhança limitada de $x_0$, com $\overline V\subset U$, e $\psi\in\mathcal C_c^\infty(U)$ tal que $\psi\equiv1$ numa vizinhança de $\overline V$. Então $\psi u$ tem suporte compacto em $U$ e
 \begin{align*}
  P(D)(\psi u)
  &=\sum_{|\alpha|\leq m}a_\alpha\partial^\alpha(\psi u)\\
  &=\psi P(D)u+
  \sum_{|\alpha|\leq m}a_\alpha
  \sum_{\substack{\beta+\gamma=\alpha\\\gamma\neq0}}
  \frac{\alpha!}{\beta!\gamma!}
  (\partial^\gamma\psi)(\partial^\beta u)\\
  &=\psi f+v.
 \end{align*}
 Aqui $v\in\mathcal E'(U)$ e $v=0$ numa vizinhança de $\overline V$, pois todas as derivadas positivas de $\psi$ se anulam onde $\psi$ é constante.

 Pela unicidade para soluções de suporte compacto,
 \[
  \psi u=E*(\psi f+v)=E*(\psi f)+E*v.
 \]
 O primeiro termo é $\mathcal C^\infty$, pois $\psi f\in\mathcal C_c^\infty$.

 Resta mostrar que $E*v$ é suave numa vizinhança de $x_0$. Escolha $\varepsilon>0$ pequeno e uma função $h_\varepsilon\in\mathcal C_c^\infty(B(0,\varepsilon))$ tal que $h_\varepsilon\equiv1$ em $B(0,\varepsilon/2)$. Escreva
 \[
  E=(1-h_\varepsilon)E+h_\varepsilon E.
 \]
 Como $E$ é suave fora da origem,
 \[
  (1-h_\varepsilon)E\in\mathcal C^\infty(\mathbb R^n).
 \]
 Logo
 \[
  [(1-h_\varepsilon)E]*v\in\mathcal C^\infty(\mathbb R^n).
 \]

 Para o segundo termo, usando a inclusão de suportes,
 \[
  \operatorname{supp}[(h_\varepsilon E)*v]
  \subseteq\operatorname{supp}(h_\varepsilon E)+\operatorname{supp}(v)
  \subseteq\overline{B(0,\varepsilon)}+\operatorname{supp}(v).
 \]
 Como $\operatorname{supp}(v)\cap\overline V=\emptyset$, podemos escolher $\varepsilon$ tão pequeno que essa soma ainda seja disjunta de uma vizinhança $W\Subset V$ de $x_0$. Assim
 \[
  (h_\varepsilon E)*v=0\quad\text{em }W.
 \]
 Consequentemente, em $W$,
 \[
  E*v=[(1-h_\varepsilon)E]*v\in\mathcal C^\infty(W).
 \]
 Como $\psi=1$ em $W$, temos $u=\psi u\in\mathcal C^\infty(W)$. Como $x_0$ foi arbitrário, $u\in\mathcal C^\infty(U)$, completando a prova. \qedsymbol
\end{proof*}
